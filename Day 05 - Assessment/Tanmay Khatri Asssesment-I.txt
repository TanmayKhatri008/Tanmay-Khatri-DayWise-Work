List is an mutable and ordered collection of data . It also allows duplicate values.It can be used to store list of item as there could be multiple same items.
Tuple is an immutable and ordered collection of data. It is used to store fixed values , which need not to be changed in future like a location on a map.
Set is an unordered collection of unique values. No repeating values allowed and no fixed order. It usually used to prefer when we have to remove the duplicates from list like to find out unique items in the store.
Dictionary is used to store key- value pairs. when we have to store multiple details of single identity like storing data of an employee or an student at that place we can use dictionary.

import { useState } from "react";
function Counter() {
  const [count, newCount] = useState(0);
  const increase = () => {
    newCount(count + 1);
  };


  const decrease = () => {
    if (count > 0) {
      newCount(count - 1);
    }
  };

  const reset = () => {
    newCount(0);
  };

  return (
    <div>
      <h2>{count}</h2>
      <button onClick={increase}>+1</button>
      <button onClick={decrease}>-1</button>
      <button onClick={reset}>Reset</button>
    </div>
  );
}

export default Counter;

import { useState} from "react";
import {useEffect} from "react";

function SearchableList() {
  const [users, setUsers] = useState([]);
  const [search, setSearch] = useState("");
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    fetch("https://jsonplaceholder.typicode.com/users")
      .then((res) => res.json())
      .then((data) => {
        setUsers(data);
        setLoading(false);
      })
      .catch(() => {
        setError("Failed to load users");
        setLoading(false);
      });
  }, []);

  const filteredUsers = users.filter((user) =>
    user.name.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <div>
      {loading && <h2>Loading...</h2>}
      {error && <h2>{error}</h2>}

      {!loading && !error && (
        <>
          <input
            type="text"
            placeholder="Search..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}/>

          {filteredUsers.length === 0 ? (
            <p>No users found</p>
          ) : (
            <ul>
              {filteredUsers.map((user) => (
                <li key={user.id}>{user.name}</li>
              ))}
            </ul>
          )}
        </>
      )}</div>
  );
}
export default SearchableList;



App.jsx

import SearchableList from "./SearchableList";
function App() {
  return (
    <div>
      <h1>Users List</h1>
      <SearchableList />
    </div>
  );
}
export default App;
